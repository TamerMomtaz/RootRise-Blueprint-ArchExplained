# 1. Template Overview

## 1.1 Purpose

These templates define the visual output format for RootRise diagnostic reports generated by **Tufte** (The Visualizer) in coordination with the full Pantheon of 11 agents. Reports embody the **&I Philosophy** by clearly distinguishing AI-generated insights from human-provided context.

## 1.2 Report Types (v2.0)

| Type | Pages | Use Case | Generation Time | Lens Requirement |
|------|-------|----------|-----------------|------------------|
| Executive Summary | 2-3 | Quick overview, investor pitches | ~30 seconds | Any |
| Full Diagnostic | 12-18 | Comprehensive analysis | ~2 minutes | Any |
| **Crema Quick Wins** | 3-4 | 30/60/90 day actionable items | ~45 seconds | EYE-CREMA active |
| **Lens-Focused** | 6-10 | Deep dive on selected lens | ~90 seconds | Single lens |
| Sector Benchmarking | 4-6 | Industry comparison focus | ~1 minute | Any |
| Action Plan | 3-4 | Implementation roadmap | ~45 seconds | Any |
| **Investor Ready** | 5-7 | Due diligence package | ~90 seconds | EYE-002 |
| **Export Roadmap** | 5-7 | Market entry timeline | ~90 seconds | EYE-001 |

## 1.3 &I Philosophy in Reports

Every report must clearly communicate:

| Element | How It's Shown | Visual Treatment |
|---------|----------------|------------------|
| AI Confidence | Color-coded confidence badges | Green/Yellow/Red indicators |
| Human Context | Highlighted callout boxes | Bronze left border, cream background |
| Data Sources | Attribution footnotes | Small text with source links |
| Uncertainty | Explicit acknowledgment | Italic text, confidence bars |
| Lens Focus | Header badge showing selected lens | Teal accent with lens icon |
| Agent Attribution | Which agent generated each insight | Agent icon + name in finding cards |

## 2.3 Score & Confidence Classification

```typescript
// Score class helper
function getScoreClass(score: number): string {
  if (score >= 80) return 'excellent';
  if (score >= 65) return 'good';
  if (score >= 50) return 'average';
  return 'needs-work';
}

function getScoreLabel(score: number, lang: string = 'en'): string {
  const labels = {
    en: { excellent: 'Excellent', good: 'Good', average: 'Average', needsWork: 'Needs Work' },
    ar: { excellent: 'ممتاز', good: 'جيد', average: 'متوسط', needsWork: 'يحتاج تحسين' }
  };
  const l = labels[lang] || labels.en;
  if (score >= 80) return l.excellent;
  if (score >= 65) return l.good;
  if (score >= 50) return l.average;
  return l.needsWork;
}

// Confidence class helper  
function getConfidenceClass(confidence: number): string {
  if (confidence >= 80) return 'high';
  if (confidence >= 60) return 'medium';
  return 'low';
}

// Effort level helper (for Crema)
function getEffortClass(effort: string): string {
  return effort; // 'low', 'medium', 'high'
}

// Timeline bucket helper (for Crema)
function getTimelineBucket(days: number): string {
  if (days <= 30) return '30';
  if (days <= 60) return '60';
  return '90';
}
```

## 9.2 Report Generation API

```typescript
import { Router, BackgroundTasks } from 'fastapi';
import { v4 as uuid } from 'uuid';

const router = new Router();

interface ReportRequest {
  session_id: string;
  report_type: ReportType;
  language: 'en' | 'ar' | 'bilingual';
  include_appendices?: boolean;
}

router.post('/reports/generate', async (request: ReportRequest, background: BackgroundTasks) => {
  const report_id = uuid();
  
  // Validate lens requirements
  const config = REPORT_CONFIGS[request.report_type];
  if (config.lens_required) {
    const session = await getSession(request.session_id);
    if (session.primary_lens !== config.lens_required) {
      return { error: `Report type ${request.report_type} requires lens ${config.lens_required}` };
    }
  }
  
  if (config.crema_required) {
    const session = await getSession(request.session_id);
    if (!session.crema_active) {
      return { error: `Report type ${request.report_type} requires The Crema to be active` };
    }
  }
  
  // Queue generation
  background.add_task(
    generateReportTask,
    report_id,
    request.session_id,
    request.report_type,
    request.language
  );
  
  return {
    report_id,
    status: 'generating',
    estimated_time_seconds: config.generation_time_seconds
  };
});

router.get('/reports/:report_id/status', async (report_id: string) => {
  const status = await redis.get(`report:${report_id}:status`);
  
  if (status === 'completed') {
    return {
      status: 'completed',
      download_url: `/api/reports/${report_id}/download`,
      preview_url: `/api/reports/${report_id}/preview`
    };
  }
  
  return { status: status || 'generating' };
});
```

## 9.3 Template Engine Setup

```typescript
import { Environment, FileSystemLoader } from 'nunjucks';

function setupTemplateEngine() {
  const env = new Environment(
    new FileSystemLoader('templates/reports'),
    { trimBlocks: true, lstripBlocks: true }
  );
  
  // Score class filter
  env.addFilter('score_class', (score: number) => {
    if (score >= 80) return 'excellent';
    if (score >= 65) return 'good';
    if (score >= 50) return 'average';
    return 'needs-work';
  });
  
  // Confidence class filter
  env.addFilter('confidence_class', (confidence: number) => {
    if (confidence >= 80) return 'high';
    if (confidence >= 60) return 'medium';
    return 'low';
  });
  
  // Effort class filter
  env.addFilter('effort_class', (effort: string) => effort);
  
  // Index plus one helper
  env.addFilter('index_plus_one', (index: number) => index + 1);
  
  // Agent lookup
  env.addFilter('agent_info', (agent_id: string) => AGENT_PROFILES[agent_id]);
  
  // Lens lookup
  env.addFilter('lens_info', (lens_code: string) => LENS_CONFIGS[lens_code]);
  
  return env;
}
```
